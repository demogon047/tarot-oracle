// FISHER-YATES SHUFFLE
// ═══════════════════════════════════════════════════════════════
function fisherYates(arr){
  const a=[...arr];
  for(let i=a.length-1;i>0;i--){
    const j=Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
  }
  return a;
}

function cutDeck(deck){
  const n=deck.length;
  const cut=Math.floor(n*.2+Math.random()*n*.6);
  return [...deck.slice(cut),...deck.slice(0,cut)];
}

function drawCards(deck,count){
  return deck.slice(0,count).map(c=>({...c,reversed:Math.random()<0.28}));
}

// STARS + NEBULA
// ═══════════════════════════════════════════════════════════════
function initStars(){
  const c=document.getElementById('stars');
  const n=Math.min(160,Math.floor(window.innerWidth*window.innerHeight/4800));
  for(let i=0;i<n;i++){
    const s=document.createElement('div');
    s.className='star'+(Math.random()<.08?' big':'');
    const sz=Math.random()*2.2+.4;
    s.style.cssText=`left:${Math.random()*100}%;top:${Math.random()*100}%;width:${sz}px;height:${sz}px;--d:${2+Math.random()*5}s;--dl:${Math.random()*6}s;--lo:${Math.random()*.1};--hi:${.3+Math.random()*.7};`;
    c.appendChild(s);
  }
  // Nebulae
  const nbs=[
    {x:15,y:20,w:300,h:200,color:'rgba(107,63,160,.08)'},
    {x:70,y:60,w:250,h:180,color:'rgba(201,168,76,.05)'},
    {x:45,y:80,w:200,h:150,color:'rgba(60,120,200,.06)'}
  ];
  nbs.forEach(nb=>{
    const d=document.createElement('div');
    d.className='nebula';
    d.style.cssText=`left:${nb.x}%;top:${nb.y}%;width:${nb.w}px;height:${nb.h}px;background:${nb.color};`;
    c.appendChild(d);
  });
}

// BUILD CARD ELEMENT
// ═══════════════════════════════════════════════════════════════
function mkCardSlot(card,posIdx,posInfo,smallMode){
  const wrap=document.createElement('div');
  wrap.className='card-wrap';
  wrap.style.animationDelay=(posIdx*.08)+'s';

  const slot=document.createElement('div');
  slot.className='card-slot';
  slot.dataset.idx=posIdx;
  if(smallMode){slot.style.cssText=`width:${parseInt(getComputedStyle(document.documentElement).getPropertyValue('--cw'))-16}px;height:${parseInt(getComputedStyle(document.documentElement).getPropertyValue('--ch'))-24}px;`;}

  const inner=document.createElement('div');
  inner.className='card-inner';

  // Back
  const back=document.createElement('div');
  back.className='card-back';
  back.innerHTML=`<div class="cb-inner"><div class="cb-bg"></div><div class="cb-star">✦</div><div class="cb-corner tl"></div><div class="cb-corner tr"></div><div class="cb-corner bl"></div><div class="cb-corner br"></div></div>`;

  // Front
  const suitClass={major:'cf-major',wands:'cf-wands',cups:'cf-cups',swords:'cf-swords',pentacles:'cf-pentacles'}[card.suit];
  const front=document.createElement('div');
  front.className=`card-front ${suitClass}${card.reversed?' is-reversed':''}`;
  if(smallMode){front.style.cssText=`width:${parseInt(getComputedStyle(document.documentElement).getPropertyValue('--cw'))-16}px;height:${parseInt(getComputedStyle(document.documentElement).getPropertyValue('--ch'))-24}px;`;}
  front.innerHTML=`
    <div class="cf-num">${card.num}</div>
    <div class="cf-icon">${card.icon}</div>
    <div class="cf-name">${card.name}</div>
    <div class="cf-nameth">${card.nameTH}</div>
    ${card.reversed?'<div class="cf-rev">↕ กลับด้าน</div>':''}
  `;

  inner.appendChild(back);
  inner.appendChild(front);
  slot.appendChild(inner);
  slot.addEventListener('click',()=>revealCard(slot,posIdx));
  wrap.appendChild(slot);

  if(posInfo){
    const lbl=document.createElement('div');
    lbl.className='pos-label';
    lbl.textContent=posInfo.name;
    const sub=document.createElement('div');
    sub.className='pos-sub';
    sub.textContent=posInfo.desc;
    wrap.appendChild(lbl);
    wrap.appendChild(sub);
  }
  return {wrap,slot};
}

// ═══════════════════════════════════════════════════════════════
// REVEAL
// ═══════════════════════════════════════════════════════════════
function revealCard(slot,idx){
  if(slot.classList.contains('revealed'))return;
  slot.classList.add('revealed');
  state.revealedCount++;
  const total=SPREADS[state.spread].positions.length;
  if(state.revealedCount>=total){
    setTimeout(()=>document.getElementById('btn-interpret').classList.remove('hidden'),600);
  }
}

// ═══════════════════════════════════════════════════════════════
// RENDER SPREAD
// ═══════════════════════════════════════════════════════════════
function renderSpread(){
  const area=document.getElementById('cards-area');
  area.innerHTML='';
  const sp=state.spread;
  const pos=SPREADS[sp].positions;
  const cards=state.drawnCards;

  if(sp===1){
    area.className='spread-1';
    const {wrap}=mkCardSlot(cards[0],0,pos[0]);
    area.appendChild(wrap);
  } else if(sp===3){
    area.className='spread-3';
    cards.forEach((c,i)=>{
      const {wrap}=mkCardSlot(c,i,pos[i]);
      area.appendChild(wrap);
    });
  } else {
    renderCelticCross(area,cards,pos);
  }
}

// Celtic Cross layout:
//       col1   col2      col3   col4  col5
// row1:         [5]             [10]
// row2: [4]  [1] [2cross] [6]  [9]
// row3:         [3]             [8]
// row4:                         [7]
function renderCelticCross(area,cards,pos){
  area.className='';
  area.style.cssText='padding:8px 0;';

  const wrapper=document.createElement('div');
  wrapper.className='celtic-grid';

  // Left cross
  const left=document.createElement('div');
  left.className='celtic-left';

  // Right staff
  const right=document.createElement('div');
  right.className='celtic-right';// helper function สร้างช่องว่าง
  function emptyCell(){
    const d=document.createElement('div');
    d.style.width='100%';
    d.style.height='100%';
    return d;
  }

  // map index ไพ่ตามตำแหน่ง Celtic Cross
  const slots = [];

  // row1
  slots.push(emptyCell()); // col1
  slots.push(mkCardSlot(cards[4],4,pos[4],true).wrap); // [5]
  slots.push(emptyCell()); // col3

  // row2
  slots.push(mkCardSlot(cards[3],3,pos[3],true).wrap); // [4]

  // center (1 + 2 ซ้อน)
  const centerWrap=document.createElement('div');
  centerWrap.style.position='relative';

  const main=mkCardSlot(cards[0],0,pos[0],true).wrap;
  const cross=mkCardSlot(cards[1],1,pos[1],true).wrap;

  cross.style.position='absolute';
  cross.style.transform='rotate(90deg)';
  cross.style.top='0';
  cross.style.left='0';

  centerWrap.appendChild(main);
  centerWrap.appendChild(cross);

  slots.push(centerWrap);

  slots.push(mkCardSlot(cards[5],5,pos[5],true).wrap); // [6]

  // row3
  slots.push(emptyCell());
  slots.push(mkCardSlot(cards[2],2,pos[2],true).wrap); // [3]
  slots.push(emptyCell());

  // append left grid
  slots.forEach(el=>left.appendChild(el));

  // RIGHT SIDE (7-10)
  [6,7,8,9].forEach(i=>{
    const {wrap}=mkCardSlot(cards[i],i,pos[i],true);
    right.appendChild(wrap);
  });

  wrapper.appendChild(left);
  wrapper.appendChild(right);
  area.appendChild(wrapper);
}
